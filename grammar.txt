Source = { declaration };

declaration       = binding | Rule ;
binding           = ident "=" "@import" "(" string ")" ";";
Rule              = ident "=" production ";";

// A Rule whose ident starts with an uppercase letter is exported and reachable
// from an importing namespace. A lowercase ident declares a local Rule, shared
// by every file of its own namespace. An ident that is "_" followed by a letter
// is a private Rule, visible only in the file that declares it. Visibility is a
// semantic rule, not a syntactic one, so each kind is spelled exactly like any
// other Rule here.

production  = sequence | alternative;
alternative = sequence { "|" production };
// A tag annotates the sequence it closes rather than standing on its own, so
// `a b c `t`` is a tagged `a b c` and not a tagged `c`. A production carries one
// tag at most, since it is annotated once. Since a tag annotates a whole
// production, a group, a repetition or an optional holds no tag of its own:
// write `( a b ) `t`` rather than `( a b `t` )`.
sequence    = basic { basic } [ tag ];
basic       = group
            | repetition
            | optional
            | term
            ;

// basic, terminal, non_terminal and term should not
// appear in the syntax tree.
group          = "(" production ")";
repetition     = "{" production "}";
optional       = "[" production "]";
term           = terminal | non_teminal;
non_terminal   = ident | member_access;
terminal       = string | regexp ;

// member_access is left-associative
member_access    = ident "." ident { "." ident };

string = '[^"\n]*';
ident  = /[a-zA-Z_][a-zA-Z0-9_]*/;
regexp = '\'[^\']*\'';
external_value = '__[a-zA-Z_][a-zA-Z0-9_]*';

// A tag annotates the element it follows, so it may only close a sequence. Its
// value is a string literal, which is why a `;` or a quote inside one is not a
// separator. A tag may not span a line, for the same reason a string may not.
tag        = '`' tag_pair { ";" tag_pair } '`';
tag_pair   = ident "=" string;
